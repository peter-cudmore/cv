\section*{Professional and Academic Experience}
\subsection*{\emph{Founder \& Chief Technology Officer} at \emph{Pneumatica Bio}, Melbourne, Australia (Aug 2025 - present).}
\begin{itemize}
	\item Designed and implemented modelling and simulation software.
	\item Built prototype liquid-handling robotic systems.
\end{itemize}

\subsection*{\emph{Founder} at \emph{VRTX Systems}, Melbourne, Australia (Mar 2024 - present).}
\begin{itemize}
	\item Embedded software-system design and implementation for autonomous systems.
	\item Designed and implemented a prototype embedded-radio stack for VRTX RDIO, including LoRa broadcast communication, secure node joining and leaving, AES encryption, and partial network-clock synchronisation.
\end{itemize}

\subsection*{\emph{Software Architect \& Algorithms Engineer} at \emph{Arkeus}, Melbourne, Australia (Nov 2022 - Mar 2024).}
\begin{itemize}
	\item Designed the software architecture and implemented software for the HSOR product.
\end{itemize}
